add_subdirectory(modules/cck-steno)
target_include_directories(app PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(CONFIG_CORNIX_STENO OR CONFIG_CCK_WHEEL_REMAP)
  if(CONFIG_ZMK_SPLIT AND NOT CONFIG_ZMK_SPLIT_ROLE_CENTRAL)
    message(FATAL_ERROR "CCK runtime must be on the central (RIGHT) only")
  endif()
  file(READ "${APPLICATION_SOURCE_DIR}/include/zmk/keymap.h" _cck_keymap_h)
  string(REGEX MATCH [[int[ \t\r\n]+zmk_keymap_layer_activate[ \t\r\n]*\([^;]+\)]] _cck_activate "${_cck_keymap_h}")
  string(REGEX MATCH [[int[ \t\r\n]+zmk_keymap_layer_deactivate[ \t\r\n]*\([^;]+\)]] _cck_deactivate "${_cck_keymap_h}")
  if(NOT _cck_activate OR NOT _cck_deactivate)
    message(FATAL_ERROR "Unsupported ZMK layer API")
  endif()
  if(_cck_activate MATCHES "," AND _cck_deactivate MATCHES ",")
    target_compile_definitions(app PRIVATE CCK_LAYER_API_ARITY=2)
  elseif(NOT _cck_activate MATCHES "," AND NOT _cck_deactivate MATCHES ",")
    target_compile_definitions(app PRIVATE CCK_LAYER_API_ARITY=1)
  else()
    message(FATAL_ERROR "Mixed ZMK layer API")
  endif()
  foreach(_fn zmk_keymap_get_layer_binding_at_idx zmk_keymap_layer_index_to_id zmk_keymap_layer_default)
    if(NOT _cck_keymap_h MATCHES "${_fn}")
      message(FATAL_ERROR "ZMK keymap header missing ${_fn}")
    endif()
  endforeach()
  target_sources(app PRIVATE src/build_contract.c)
endif()
if(CONFIG_CCK_WHEEL_REMAP)
  target_sources(app PRIVATE src/behavior_cck_wheel.c)
endif()
if(CONFIG_CCK_USB_AUTO_SELECT)
  file(READ "${APPLICATION_SOURCE_DIR}/include/zmk/endpoints.h" _cck_ep_h)
  if(_cck_ep_h MATCHES "zmk_endpoint_set_preferred_transport")
    target_compile_definitions(app PRIVATE CORNIX_SELECT_USB_TRANSPORT=zmk_endpoint_set_preferred_transport)
  elseif(_cck_ep_h MATCHES "zmk_endpoints_select_transport")
    target_compile_definitions(app PRIVATE CORNIX_SELECT_USB_TRANSPORT=zmk_endpoints_select_transport)
  else()
    message(FATAL_ERROR "Unsupported ZMK endpoint API")
  endif()
  target_sources(app PRIVATE src/usb_auto_select.c)
endif()
# Record what mutable original west references actually resolved to in CI logs.
# Failure to read git metadata here is not substituted with a made-up revision.
execute_process(COMMAND git -C "${APPLICATION_SOURCE_DIR}" rev-parse HEAD
  RESULT_VARIABLE _cck_git_rc OUTPUT_VARIABLE _cck_git_sha
  OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(_cck_git_rc EQUAL 0)
  message(STATUS "CCK resolved ZMK commit: ${_cck_git_sha}")
endif()
if(EXISTS "${ZEPHYR_BASE}/../zmk-pmw3610-driver")
  execute_process(COMMAND git -C "${ZEPHYR_BASE}/../zmk-pmw3610-driver" rev-parse HEAD
    RESULT_VARIABLE _cck_drv_rc OUTPUT_VARIABLE _cck_drv_sha
    OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  if(_cck_drv_rc EQUAL 0)
    message(STATUS "CCK resolved PMW3610 driver commit: ${_cck_drv_sha}")
  endif()
endif()

if(CONFIG_CCK_STENO_POINTER_ASSIST)
  if(NOT CONFIG_CORNIX_STENO OR NOT CONFIG_ZMK_POINTING OR
     (CONFIG_ZMK_SPLIT AND NOT CONFIG_ZMK_SPLIT_ROLE_CENTRAL))
    message(FATAL_ERROR "CCK pointer assist requires the STENO/pointing central")
  endif()
  set(_cck_ip_header "${APPLICATION_SOURCE_DIR}/include/drivers/input_processor.h")
  if(NOT EXISTS "${_cck_ip_header}")
    message(FATAL_ERROR "CCK EO scrolling needs ZMK input processor API")
  endif()
  file(READ "${_cck_ip_header}" _cck_ip_text)
  if(NOT _cck_ip_text MATCHES "zmk_input_processor_state" OR
     NOT _cck_ip_text MATCHES "handle_event")
    message(FATAL_ERROR "Unsupported ZMK input processor API")
  endif()
  target_sources(app PRIVATE src/cck_steno_mouse.c src/input_processor_cck_scroll.c)
endif()
